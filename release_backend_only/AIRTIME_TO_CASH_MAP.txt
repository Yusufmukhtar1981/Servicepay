   78:   Future<String?> getSavedAuthToken(
   79:     SharedPreferences preferences,
   80:   ) async {
   81:     const List<String> tokenKeys = <String>[
   82:       'auth_token',
   83:       'token',
   84:       'access_token',
   85:       'accessToken',
   86:       'jwt_token',
   87:       'jwt',
   88:     ];
   89: 
   90:     for (final String key in tokenKeys) {
   91:       final String? value = preferences.getString(key);
   92: 
   93:       if (value == null || value.trim().isEmpty) {
   94:         continue;
   95:       }
   96: 
   97:       String token = value.trim();
   98: 
   99:       if (token.toLowerCase().startsWith('bearer ')) {
  100:         token = token.substring(7).trim();
  101:       }
  102: 
  103:       if (token.isNotEmpty) {
  104:         return token;
  105:       }
  106:     }
  107: 
  108:     return null;
  109:   }

  111:   Future<void> loadDashboard({
  112:     bool refreshing = false,
  113:   }) async {
  114:     if (mounted) {
  115:       setState(() {
  116:         if (refreshing) {
  117:           isRefreshing = true;
  118:         } else {
  119:           isLoading = true;
  120:         }
  121:       });
  122:     }
  123: 
  124:     try {
  125:       final SharedPreferences preferences =
  126:           await SharedPreferences.getInstance();
  127: 
  128:       final String savedName = preferences.getString('user_name') ??
  129:           preferences.getString('full_name') ??
  130:           preferences.getString('name') ??
  131:           'Customer';
  132: 
  133:       final double savedBalance = preferences.getDouble('wallet_balance') ?? 0;
  134: 
  135:       if (mounted) {
  136:         setState(() {
  137:           userName = savedName.trim().isEmpty ? 'Customer' : savedName.trim();
  138: 
  139:           walletBalance = savedBalance;
  140:         });
  141:       }
  142: 
  143:       await _loadServiceAvailability();
  144: 
  145:       final String? token = await getSavedAuthToken(preferences);
  146: 
  147:       if (token == null || token.isEmpty) {
  148:         return;
  149:       }
  150: 
  151:       final http.Response response = await http.get(
  152:         Uri.parse('$baseUrl/wallet'),
  153:         headers: <String, String>{
  154:           'Accept': 'application/json',
  155:           'Authorization': 'Bearer $token',
  156:         },
  157:       ).timeout(
  158:         const Duration(seconds: 30),
  159:       );
  160: 
  161:       if (response.statusCode != 200) {
  162:         return;
  163:       }
  164: 
  165:       final dynamic decoded = jsonDecode(response.body);
  166: 
  167:       if (decoded is! Map) {
  168:         return;
  169:       }
  170: 
  171:       final Map<String, dynamic> data = Map<String, dynamic>.from(decoded);
  172: 
  173:       dynamic rawBalance = data['walletBalance'] ?? data['balance'];
  174: 
  175:       if (data['data'] is Map) {
  176:         final Map<String, dynamic> nested = Map<String, dynamic>.from(
  177:           data['data'] as Map,
  178:         );
  179: 
  180:         rawBalance ??= nested['walletBalance'] ?? nested['balance'];
  181:       }
  182: 
  183:       final double freshBalance = rawBalance is num
  184:           ? rawBalance.toDouble()
  185:           : double.tryParse(
  186:                 rawBalance?.toString() ?? '',
  187:               ) ??
  188:               walletBalance;
  189: 
  190:       await preferences.setDouble(
  191:         'wallet_balance',
  192:         freshBalance,
  193:       );
  194: 
  195:       if (mounted) {
  196:         setState(() {
  197:           walletBalance = freshBalance;
  198:         });
  199:       }
  200:     } catch (_) {
  201:       // Keep locally saved dashboard values.
  202:     } finally {
  203:       if (mounted) {
  204:         setState(() {
  205:           isLoading = false;
  206:           isRefreshing = false;
  207:         });
  208:       }
  209:     }
  210:   }
